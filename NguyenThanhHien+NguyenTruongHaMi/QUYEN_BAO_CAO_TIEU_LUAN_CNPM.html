CREATE TABLE Sach (
    MaSach VARCHAR(20) PRIMARY KEY,
    TenSach NVARCHAR(250) NOT NULL,
    TheLoai CHAR(1) NOT NULL CHECK (TheLoai IN ('A', 'B', 'C')),
    TacGia NVARCHAR(150) NOT NULL,
    NamXuatBan INT NOT NULL,
    NhaXuatBan NVARCHAR(150) NOT NULL,
    NgayNhap DATE NOT NULL,
    TriGia DECIMAL(18,0) NOT NULL CHECK (TriGia >= 0),
    NguoiTiepNhan VARCHAR(20) NOT NULL REFERENCES NhanVien(MaNV),
    TinhTrang VARCHAR(20) NOT NULL DEFAULT 'Con' CHECK (TinhTrang IN ('Con', 'DangMuon')),
    HinhAnh NVARCHAR(500)
);

    return `<tr><td>${idx + 1}</td><td>${b.maSach}</td><td>${b.tenSach}</td><td>Loại ${b.theLoai}</td><td>${b.tacGia}</td></tr>`;
  }).join('');
}

C. PHẦN KẾT LUẬN

Ý nghĩa của đề tài đối với bản thân trong quá trình học tập

Quá trình nghiên cứu, phân tích thiết kế và trực tiếp lập trình hoàn thiện Đề tài 1.1: Thiết kế Phần mềm Quản lý Thư viện trong học phần Công Nghệ Phần Mềm đã mang lại cho nhóm chúng em (Nguyễn Thanh Hiền và Nguyễn Trương Hà Mi) những kiến thức và kinh nghiệm thực hành vô cùng quý báu:

  • Làm chủ quy trình phát triển phần mềm chuẩn mực: Biết cách đi từ khâu thu thập yêu cầu nghiệp vụ (BM1 → BM4, QĐ1 → QĐ4), xây dựng mô hình PTTKHT (DFD, ERD) và PTTKĐT (Use Case, Class, Sequence Diagram) đến thiết kế cơ sở dữ liệu SQL Server đạt chuẩn 3NF.
  • Chuyển hóa quy định nghiệp vụ thành mã lệnh kiểm soát tự động: Biết cách kết hợp kiểm tra hợp lệ ở cả tầng Giao diện (JavaScript Validation) và tầng Cơ sở dữ liệu (SQL CHECK CONSTRAINT & TRIGGER) để ngăn chặn triệt để dữ liệu sai quy định.
  • Nâng cao tư duy thiết kế UI/UX thực tế: Xây dựng được phần mềm có giao diện trực quan, tách biệt rõ ràng giữa Cổng Độc giả OPAC bên ngoài và Cổng Cán bộ Quản trị nội bộ bên trong, tạo nền tảng vững chắc cho các đồ án chuyên ngành và công việc phát triển phần mềm sau này.
